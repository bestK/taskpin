# Compilation settings that should be applied to most targets.
#
# Be cautious about adding new options here, as plugins use this function by
# default. In most cases, you should add new options to specific targets instead
# of modifying this function.
function(APPLY_STANDARD_SETTINGS TARGET)
  target_compile_features(${TARGET} PUBLIC cxx_std_17)
  target_compile_options(${TARGET} PRIVATE /W4 /WX /wd"4100")
  target_compile_options(${TARGET} PRIVATE /EHsc)
  # /utf-8 tells MSVC the sources are UTF-8 for both reading and execution.
  # Without it MSVC assumes the local ANSI codepage — 936 on a zh-CN box — and
  # emits C4819 on every file carrying a non-ASCII character. This project's
  # comments are partly Chinese, and /WX turns that warning into an error, so
  # the build fails outright:
  #
  #   main.cpp(1,1): error C2220: the following warning is treated as an error
  #   main.cpp(1,1): warning C4819: the file contains a character that cannot
  #                  be represented in the current code page (936)
  #
  # Passing /utf-8 is the correct fix rather than transliterating the comments:
  # the sources genuinely are UTF-8, and this is also what Flutter's own
  # Windows template does for exactly this reason.
  target_compile_options(${TARGET} PRIVATE /utf-8)
  target_compile_definitions(${TARGET} PRIVATE "_HAS_EXCEPTIONS=0")
  target_compile_definitions(${TARGET} PRIVATE "$<$<CONFIG:Debug>:_DEBUG>")
endfunction()

# Flutter library and tool build rules.

add_subdirectory(${FLUTTER_MANAGED_DIR})

# Application build; see runner/CMakeLists.txt.
add_subdirectory("runner")

# Generated plugin build rules, which manage building the plugins and adding
# them to the application.
#
# NOTE: this deliberately includes `generated_plugins_local.cmake` (ours) rather
# than `flutter/generated_plugins.cmake` (Flutter's). Flutter's version does
# `add_subdirectory(flutter/ephemeral/.plugin_symlinks/<plugin>/windows)`, and
# on this machine that symlink tree is created but NOT traversable, so CMake
# fails with "which is not an existing directory" before anything is built.
#
# Ours points at the real pub-cache directories from
# `.flutter-plugins-dependencies`, so no reparse point is involved. It is
# regenerated by `tool/generate_plugins_cmake.py`, which
# `tool/build_windows.py` runs automatically. See that script's docstring for
# the full diagnosis.
#
# Fall back to Flutter's file when ours is absent, so a fresh clone that has
# only run `flutter pub get` still configures.
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/generated_plugins_local.cmake")
  include(generated_plugins_local.cmake)
else()
  include(flutter/generated_plugins.cmake)
endif()

# === TaskPin native libraries ===
#
# The Lua engine (native/engine) and the native taskbar bar (native/win) are
# built as STATIC libraries and linked straight into the runner, so the shipped
# app stays a single binary with no runtime DLLs to carry. See
# docs/FLUTTER_REWRITE_PLAN.md, "C 库的构建方式".
#
# Both trees are owned by other agents; everything they need is arranged from
# here rather than by editing their CMakeLists.

# native/engine also builds tool/engine_cli.c into a standalone `engine_cli`
# harness. That harness is how Phase 2 captures its golden JSON, so it must keep
# working when built directly — but it has no business in the app binary.
# TP_BUILD_CLI defaults ON, so it is forced off for THIS subdirectory only.
set(TP_BUILD_CLI OFF CACHE BOOL "" FORCE)
add_subdirectory(
  "${CMAKE_CURRENT_SOURCE_DIR}/../native/engine"
  "${CMAKE_BINARY_DIR}/taskpin_engine"
)

# native/win: the Direct2D bar surface plus the window-embedding helpers.
add_subdirectory(
  "${CMAKE_CURRENT_SOURCE_DIR}/../native/win"
  "${CMAKE_BINARY_DIR}/taskpin_bar"
)

# native/win also defines `taskpin_bar_demo`, a console harness with its own
# wmain(). The `-municode` link flag it needs is guarded by `if(MINGW)` in that
# CMakeLists, so under MSVC it would fail to link with "unresolved external
# symbol main". EXCLUDE_FROM_ALL keeps it out of the default build without
# touching a directory this phase does not own; `taskpin_bar` itself is a
# normal target and is built and linked as usual.
if(TARGET taskpin_bar_demo)
  set_target_properties(taskpin_bar_demo PROPERTIES EXCLUDE_FROM_ALL TRUE)
endif()

# The bar library is C++17 and is compiled with the runner's MSVC toolchain, so
# it inherits UNICODE/_UNICODE from the add_definitions() above.
#
# The engine must NOT: `native/engine/tp_http_win.c` states in its header that
# "UNICODE is NOT defined for this engine (the ABI is UTF-8)". The top-level
# add_definitions() would otherwise leak those macros in and silently switch any
# TCHAR-macro'd Win32 call to its -W variant. Removing them here keeps the
# engine compiling exactly as its own standalone build does, without editing a
# directory this phase does not own.
foreach(tp_engine_target taskpin_engine taskpin_lua)
  if(TARGET ${tp_engine_target})
    get_target_property(tp_defs ${tp_engine_target} COMPILE_DEFINITIONS)
    if(tp_defs)
      list(REMOVE_ITEM tp_defs UNICODE _UNICODE)
      set_target_properties(${tp_engine_target} PROPERTIES
        COMPILE_DEFINITIONS "${tp_defs}")
    endif()
  endif()
endforeach()

target_link_libraries(${BINARY_NAME} PRIVATE taskpin_engine taskpin_bar)


# === Installation ===
# Support files are copied into place next to the executable, so that it can
# run in place. This is done instead of making a separate bundle (as on Linux)
# so that building and running from within Visual Studio will work.
set(BUILD_BUNDLE_DIR "$<TARGET_FILE_DIR:${BINARY_NAME}>")
# Make the "install" step default, as it's required to run.
set(CMAKE_VS_INCLUDE_INSTALL_TO_DEFAULT_BUILD 1)
if(CMAKE_INSTALL_PREFIX_INITIALIZED_TO_DEFAULT)
  set(CMAKE_INSTALL_PREFIX "${BUILD_BUNDLE_DIR}" CACHE PATH "..." FORCE)
endif()

set(INSTALL_BUNDLE_DATA_DIR "${CMAKE_INSTALL_PREFIX}/data")
set(INSTALL_BUNDLE_LIB_DIR "${CMAKE_INSTALL_PREFIX}")

install(TARGETS ${BINARY_NAME} RUNTIME DESTINATION "${CMAKE_INSTALL_PREFIX}"
  COMPONENT Runtime)

install(FILES "${FLUTTER_ICU_DATA_FILE}" DESTINATION "${INSTALL_BUNDLE_DATA_DIR}"
  COMPONENT Runtime)

install(FILES "${FLUTTER_LIBRARY}" DESTINATION "${INSTALL_BUNDLE_LIB_DIR}"
  COMPONENT Runtime)

if(PLUGIN_BUNDLED_LIBRARIES)
  install(FILES "${PLUGIN_BUNDLED_LIBRARIES}"
    DESTINATION "${INSTALL_BUNDLE_LIB_DIR}"
    COMPONENT Runtime)
endif()

# Copy the native assets provided by the build.dart from all packages.
set(NATIVE_ASSETS_DIR "${PROJECT_BUILD_DIR}native_assets/windows/")
install(DIRECTORY "${NATIVE_ASSETS_DIR}"
   DESTINATION "${INSTALL_BUNDLE_LIB_DIR}"
   COMPONENT Runtime)

# Fully re-copy the assets directory on each build to avoid having stale files
# from a previous install.
set(FLUTTER_ASSET_DIR_NAME "flutter_assets")
install(CODE "
  file(REMOVE_RECURSE \"${INSTALL_BUNDLE_DATA_DIR}/${FLUTTER_ASSET_DIR_NAME}\")
  " COMPONENT Runtime)
install(DIRECTORY "${PROJECT_BUILD_DIR}/${FLUTTER_ASSET_DIR_NAME}"
  DESTINATION "${INSTALL_BUNDLE_DATA_DIR}" COMPONENT Runtime)

# Install the AOT library on non-Debug builds only.
install(FILES "${AOT_LIBRARY}" DESTINATION "${INSTALL_BUNDLE_DATA_DIR}"
  CONFIGURATIONS Profile;Release
  COMPONENT Runtime)
